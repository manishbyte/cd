Practical 1: Lexical Analyzer using LEX to Identify C Tokens

Step 1: Create Practical Workspace Directory

Open an Ubuntu terminal (Ctrl + Alt + T) and create a dedicated working directory on your Desktop:

Ubuntu Bash Terminal 
mkdir -p ~/Desktop/CompilerLab/Practical1
cd ~/Desktop/CompilerLab/Practical1

Step 2: Create the Sample C Source File (input.c)

Create the input file that will be fed into the lexical analyzer:

Ubuntu Command 
gedit input.c

Paste the following C source code and save the file:

input.c 
#include<stdio.h>
int add(int a, int b)
{
    int c = a + b;
    return c;
}
void main()
{
    int x = 5;
    /* this is a comment */
    if (x > 3 && x != 9)
    {
        printf("Hello");
    }
}

Step 3: Create the Flex Lexer Specification (a.l)

Create the Lex definition file using your text editor:

Ubuntu Command 
gedit a.l

Paste the complete Flex program below and save:

a.l 
%option noyywrap
%{
#include <stdio.h>
%}

digit       [0-9]+
header      "#include"[ \t]*"<"[a-zA-Z0-9_]+".h>"
inbuilt     (printf|scanf|clrscr|getch|exit)[ \t]*"("[^;\n]*")"
comment     "/*"([^*]|\*+[^*/])*\*+"/"
main_func   ("void"|"int")[ \t]+"main"[ \t]*"("[ \t]*")"
function    ("void"|"int"|"char"|"float"|"double")[ \t]+[a-zA-Z_][a-zA-Z0-9_]*[ \t]*"("[^)\n]*")"
datatype    ("int"|"char"|"float"|"double"|"void")[ \t]+
operator    [+\-*/%]
assign      "="
relational  ("<="|\">=\"|\"==\"|\"!=\"|\"<\"|\">\")
logical     ("&&"|"||")
terminator  ";"
bracket1    "{"
bracket2    "}"
loop        (if|else|for|while|do)
word        [a-zA-Z_][a-zA-Z0-9_]*

%%
{digit}         { printf("\n Number                  :: %s", yytext); }
{header}        { printf("\n Header File             :: %s", yytext); }
{inbuilt}       { printf("\n Function                :: %s", yytext); }
{comment}       { printf("\n Comment                 :: %s", yytext); }
{main_func}     { printf("\n Main Function           :: %s", yytext); }
{function}      { printf("\n User function           :: %s", yytext); }
{datatype}      { printf("\n Datatype                :: %s", yytext); }
{operator}      { printf("\n Operator                :: %s", yytext); }
{assign}        { printf("\n Assignment operator     :: %s", yytext); }
{relational}    { printf("\n Relational operator     :: %s", yytext); }
{logical}       { printf("\n Logical operator        :: %s", yytext); }
{terminator}    { printf("\n Terminator              :: %s", yytext); }
{bracket1}      { printf("\n Opening curly bracket   :: %s", yytext); }
{bracket2}      { printf("\n Closing curly bracket   :: %s", yytext); }
{loop}          { printf("\n Loop/Condition keyword  :: %s", yytext); }
{word}          { printf("\n Variable/Identifier     :: %s", yytext); }
[ \t\r\n]+      ;
.               ;
%%

int main(int argc, char *argv[])
{
    FILE *fp;
    if (argc < 2) {
        printf("Usage: %s input.c\n", argv[0]);
        return 1;
    }
    fp = fopen(argv[1], "r");
    if (fp == NULL) {
        printf("Error opening input file\n");
        return 1;
    }
    yyin = fp;
    yylex();
    fclose(fp);
    printf("\n");
    return 0;
}

Step 4: Compile and Execute on Ubuntu

Run the Flex generator, compile the resulting C scanner with GCC, and execute passing input.c:

Terminal Build & Run 
flex a.l
gcc lex.yy.c -o a
./a input.c
